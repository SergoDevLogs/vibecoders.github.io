#include <stdio.h>

int main() {
    int N;
    printf("Введите число элементов массива: ");
    scanf("%d", &N);
    
    if (N < 1) {
        printf("N не может быть меньше 1\n");
        return 1;
    }
    
    int mas[N];
    printf("Введите %d целых чисел: \n", N);
    for (int i = 0; i < N; i++) {
        scanf("%d", &mas[i]);
    }
    
    // Сортировка массива по убыванию (пузырьковая сортировка)
    for (int i = 0; i < N - 1; i++) {
        for (int j = 0; j < N - i - 1; j++) {
            if (mas[j] < mas[j + 1]) {
                // Обмен элементов
                int temp = mas[j];
                mas[j] = mas[j + 1];
                mas[j + 1] = temp;
            }
        }
    }
    
    printf("Массив, упорядоченный по убыванию:\n");
    for (int i = 0; i < N; i++) {
        printf("%d ", mas[i]);
    }
    printf("\n");
    
    return 0;
}
